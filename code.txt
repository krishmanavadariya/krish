import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Load data
movies = pd.read_csv("Movies.csv")
shows = pd.read_csv("Shows.csv")
bookings = pd.read_csv("Bookings.csv")

# Explore
print(movies.head())
print(movies.shape)
print(movies.info())
print(movies.describe())
print(movies.isnull().sum())
print(movies.duplicated().sum())

# Clean
movies = movies.drop_duplicates()
shows = shows.drop_duplicates()
bookings = bookings.drop_duplicates()

movies["Genre"] = movies["Genre"].fillna("Unknown")

shows["Show_Date"] = pd.to_datetime(
    shows["Show_Date"], errors="coerce"
)

bookings["Ticket_Price"] = pd.to_numeric(
    bookings["Ticket_Price"], errors="coerce"
)

bookings["Ticket_Price"] = bookings["Ticket_Price"].fillna(
    bookings["Ticket_Price"].median()
)

# Validate
print(movies.isnull().sum())
print(shows.isnull().sum())
print(bookings.isnull().sum())

print(movies.duplicated().sum())
print(shows.duplicated().sum())
print(bookings.duplicated().sum())

# Visualization
plt.hist(bookings["Ticket_Price"])
plt.xlabel("Ticket Price")
plt.ylabel("Frequency")
plt.title("Ticket Price Distribution")
plt.show()

movies["Genre"].value_counts().plot(kind="bar")
plt.xlabel("Genre")
plt.ylabel("Number of Movies")
plt.title("Movies by Genre")
plt.show()

# Export
movies.to_csv("Movies_Cleaned.csv", index=False)
shows.to_csv("Shows_Cleaned.csv", index=False)
bookings.to_csv("Bookings_Cleaned.csv", index=False)

print("Done")


| Command               | Meaning             |
| --------------------- | ------------------- |
| `pd.read_csv()`       | Load CSV            |
| `.head()`             | First 5 rows        |
| `.shape`              | Rows & columns      |
| `.info()`             | Data types/info     |
| `.describe()`         | Statistics          |
| `.isnull().sum()`     | Missing values      |
| `.duplicated().sum()` | Duplicate count     |
| `.drop_duplicates()`  | Remove duplicates   |
| `.fillna()`           | Fill missing values |
| `.to_csv()`           | Save CSV            |




1. read_csv() → 2. head() → 3. shape → 4. info() → 5. describe() → 6. isnull() → 7. drop_duplicates() → 8. fillna() → 9. Matplotlib → 10. to_csv()
